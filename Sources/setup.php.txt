// API keys are shown with visible head + tail so the operator can verify
// WHICH key is configured without exposing it fully (e.g. `sk-or-…ab12`).
// Passwords keep the full mask above.
$mask_key = static function (?string $v): string {
    $v = (string) $v;
    if ($v === '') return '';
    $len = strlen($v);
    if ($len <= 8) return str_repeat('•', max(4, $len));
    $head = substr($v, 0, min(6, $len - 4));
    $tail = substr($v, -4);
    return $head . '…' . str_repeat('•', 4) . '…' . $tail;
};
// Effective value (DB setting → else config.php resolved value).
$eff = static function (string $key, $cfgKey = null) use ($current, $cfg) {
    $cfgKey = $cfgKey ?? $key;
    if (isset($current[$key]) && $current[$key] !== '') return (string) $current[$key];
    return (string) ($cfg[$cfgKey] ?? '');
};
